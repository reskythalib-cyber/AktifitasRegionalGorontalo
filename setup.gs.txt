function setupTriggers() {
  try {
    var functionName = 'autoFillAlphaAbsensi';
    var triggers = ScriptApp.getProjectTriggers();
    var triggerExists = false;

    for (var i = 0; i < triggers.length; i++) {
      if (triggers[i].getHandlerFunction() === functionName) {
        triggerExists = true;
        break;
      }
    }

    if (!triggerExists) {
      ScriptApp.newTrigger(functionName)
        .timeBased()

/**
 * ZETTBOT FEATURE 1: Zero-ID Safety Guard
 * Mengecek apakah nilai ID valid (bukan angka 0, 00, -, strip, atau kosong).
 * Jika mengembalikan false, sistem WAJIB mengabaikan ID dan beralih ke logika Nama Fuzzy.
 *
 * @param {string|number} idVal - Nilai ID Loan atau Customer Number mentah
 * @return {boolean} True jika ID valid dan bermakna angka > 0
 */
function isValidId_(idVal) {
  if (idVal === null || idVal === undefined) return false;
  var str = String(idVal).trim().replace(/['"`\s]/g, '');
  if (!str || str === '-' || str === 'null' || str === 'undefined') return false;
  var numericOnly = str.replace(/[^0-9]/g, '');
  if (!numericOnly) return false;
  // Periksa apakah hanya terdiri dari angka 0 (misal '0', '00', '000')
  var num = parseInt(numericOnly, 10);
  return !isNaN(num) && num > 0;
}

/**
 * ZETTBOT FEATURE 2: Normalized Numeric ID Normalizer
 * Menormalisasi nilai ID Loan dan Customer Number dengan membuang
 * spasi, tanda kutip (' atau "), serta seluruh karakter non-digit.
 *
 * @param {string|number} rawId - Nilai ID mentah dari Excel atau database sheet
 * @return {string} Nilai ID bersih berupa deretan angka murni atau string kosong jika ID invalid
 */
function normalizeNumericId_(rawId) {
  if (!isValidId_(rawId)) return '';
  var cleanStr = String(rawId).trim().replace(/['"`\s]/g, '');
  return cleanStr.replace(/[^0-9]/g, '');
}

/**
 * ZETTBOT FEATURE 3: Automatic WhatsApp Link & Phone Normalizer
 * Menormalisasi nomor telepon dari input user (08..., +62..., 8...) ke format
 * internasional standar WhatsApp (628...) secara otomatis dan aman.
 *
 * @param {string|number} phone - Nomor telepon mentah dari database sheet
 * @return {string} Nomor bersih dengan format 628...
 */
function formatWhatsAppNumber(phone) {
  if (!phone && phone !== 0) return '';
  var clean = String(phone).replace(/[^0-9]/g, '').trim();
  if (!clean) return '';

  if (clean.indexOf('08') === 0) {
    clean = '62' + clean.substring(1);
  } else if (clean.indexOf('8') === 0 && clean.indexOf('628') !== 0) {
    clean = '62' + clean;
  } else if (clean.indexOf('6208') === 0) {
    clean = '62' + clean.substring(3);
  }

  return clean;
}

/**
 * ZETTBOT FEATURE 4: Generator Tautan Pesan WhatsApp Langsung
 *
 * @param {string|number} phone - Nomor telepon
 * @param {string} messageText - Teks pesan konfirmasi
 * @return {string} URL langsung https://wa.me/62...
 */
function generateWhatsAppLink(phone, messageText) {
  var waNumber = formatWhatsAppNumber(phone);
  if (!waNumber) return '';
  var textEncoded = messageText ? encodeURIComponent(messageText) : '';
  return 'https://wa.me/' + waNumber + (textEncoded ? '?text=' + textEncoded : '');
}
