/**
 * ZETTBOT FEATURE 1: Zero-ID Safety Guard
 * Menolak ID bernilai 0, 00, -, atau kosong agar tidak dianggap ID valid
 */
function isValidId_(idVal) {
  if (idVal === null || idVal === undefined) return false;
  var str = String(idVal).trim().replace(/['"`\s]/g, '');
  if (!str || str === '-' || str === 'null' || str === 'undefined' || str === '0' || str === '00' || str === '000') return false;
  var numericOnly = str.replace(/[^0-9]/g, '');
  if (!numericOnly) return false;
  var num = parseInt(numericOnly, 10);
  return !isNaN(num) && num > 0;
}

/**
 * ZETTBOT FEATURE 2: Normalized Numeric ID Normalizer
 * Menormalisasi ID Loan & Customer Number anti-spasi dan anti-karakter non-digit
 */
function normalizeNumericId_(rawId) {
  if (!isValidId_(rawId)) return '';
  var cleanStr = String(rawId).trim().replace(/['"`\s]/g, '');
  return cleanStr.replace(/[^0-9]/g, '');
}

function cleanGoogleDriveUrl_(url) {
  if (!url || typeof url !== 'string') return '';
  url = url.trim();
  if (url === '-' || url === '') return '';

  if (url.indexOf('lh3.googleusercontent.com/d/') !== -1) return url;

  if (url.indexOf('drive.google.com') !== -1 || url.indexOf('docs.google.com') !== -1) {
    var idMatch = url.match(/\/file\/d\/([a-zA-Z0-9_-]{20,})/) || 
                  url.match(/id=([a-zA-Z0-9_-]{20,})/) || 
                  url.match(/\/d\/([a-zA-Z0-9_-]{20,})/);
    if (idMatch && idMatch[1]) {
      var lower = url.toLowerCase();
      var isMedia = lower.indexOf('video') !== -1 || lower.indexOf('.mp4') !== -1 ||
                    lower.indexOf('.webm') !== -1 || lower.indexOf('.mov') !== -1 ||
                    lower.indexOf('.3gp') !== -1 || lower.indexOf('.mp3') !== -1 ||
                    lower.indexOf('.wav') !== -1 || lower.indexOf('.m4a') !== -1;
      if (isMedia) {
        return 'https://drive.google.com/file/d/' + idMatch[1] + '/preview';
      }
      return 'https://lh3.googleusercontent.com/d/' + idMatch[1];
    }
  }

  return url;
}

function parseDateAny_(str) {
  if (!str) return null;
  if (str instanceof Date) return isNaN(str.getTime()) ? null : str;
  str = String(str).trim();
  var datePart = str.split(' ')[0];

  if (datePart.indexOf('/') !== -1) {
    var p = datePart.split('/');
    if (p.length === 3) {
      var p0 = parseInt(p[0], 10);
      var p1 = parseInt(p[1], 10);
      var p2 = parseInt(p[2], 10);
      var y, m, d;
      if (p[0].length === 4) {
        y = p0; m = p1 - 1; d = p2;
      } else {
        d = p0; m = p1 - 1; y = p2;
        if (p1 > 12 && p0 <= 12) {
          m = p0 - 1; d = p1;
        }
      }
      var dt = new Date(y, m, d);
      return isNaN(dt.getTime()) ? null : dt;
    }
  } else if (datePart.indexOf('-') !== -1) {
    var p2 = datePart.split('-');
    if (p2.length === 3) {
      if (p2[0].length === 4) {
        var dt2 = new Date(parseInt(p2[0], 10), parseInt(p2[1], 10) - 1, parseInt(p2[2], 10));
        return isNaN(dt2.getTime()) ? null : dt2;
      } else {
        var dt3 = new Date(parseInt(p2[2], 10), parseInt(p2[1], 10) - 1, parseInt(p2[0], 10));
        return isNaN(dt3.getTime()) ? null : dt3;
      }
    }
  }

  var parsed = new Date(str);
  return isNaN(parsed.getTime()) ? null : parsed;
}

function isSameDay_(dateInput, targetDateObj) {
  if (!dateInput || !targetDateObj) return false;
  var d = parseDateAny_(dateInput);
  if (!d || isNaN(d.getTime())) return false;
  return d.getDate() === targetDateObj.getDate() &&
         d.getMonth() === targetDateObj.getMonth() &&
         d.getFullYear() === targetDateObj.getFullYear();
}

function getRowDateTimeMs_(row) {
  if (!row) return 0;
  var dStr = row.tanggal || row.disbursement_date || row.tanggal_approve_uk || row.waktu || row.submittedat || row.timestamp || row['_raw_12'] || row['_raw_0'] || '';
  var tStr = row.waktu || row['_raw_13'] || row['_raw_1'] || '';
  var dt = parseDateAny_(dStr);
  var ms = dt ? dt.getTime() : 0;
  if (tStr && String(tStr).indexOf(':') !== -1) {
    var tp = String(tStr).split(':');
    ms += (parseInt(tp[0], 10) || 0) * 3600000 + (parseInt(tp[1], 10) || 0) * 60000;
  }
  return ms;
}

function getSheetLastUpdated_(sheet, sheetKey, rawRows) {
  try {
    if (!sheet) return '';
    var cache = CacheService.getScriptCache();
    var cachedTime = cache.get('sheet_mod_' + sheetKey);
    if (cachedTime) return cachedTime;

    var latestDateFound = null;
    if (rawRows && rawRows.length > 0) {
      var scanCount = Math.min(rawRows.length, 120);
      for (var i = 0; i < scanCount; i++) {
        var r = rawRows[i];
        var dateCandidates = [
          r.tanggal_approve_uk, r.disbursement_date, r.tanggal,
          r.waktu, r.submittedat, r.timestamp, r['_raw_0'], r['_raw_1']
        ];
        for (var c = 0; c < dateCandidates.length; c++) {
          var val = dateCandidates[c];
          if (val) {
            var parsed = parseDateAny_(val);
            if (parsed && !isNaN(parsed.getTime())) {
              if (parsed.getFullYear() >= 2024 && parsed.getTime() <= (Date.now() + 86400000)) {
                if (!latestDateFound || parsed.getTime() > latestDateFound.getTime()) {
                  latestDateFound = parsed;
                }
              }
            }
          }
        }
      }
    }

    if (latestDateFound) {
      var res = Utilities.formatDate(latestDateFound, CONFIG.TIMEZONE, 'dd/MM/yyyy HH:mm') + ' WITA';
      try { cache.put('sheet_mod_' + sheetKey, res, CONFIG.CACHE_TTL); } catch(e){}
      return res;
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (ss) {
      var file = DriveApp.getFileById(ss.getId());
      return Utilities.formatDate(file.getLastUpdated(), CONFIG.TIMEZONE, 'dd/MM/yyyy HH:mm') + ' WITA';
    }
  } catch(e) {
    return Utilities.formatDate(new Date(), CONFIG.TIMEZONE, 'dd/MM/yyyy HH:mm') + ' WITA';
  }
  return Utilities.formatDate(new Date(), CONFIG.TIMEZONE, 'dd/MM/yyyy HH:mm') + ' WITA';
}

function findHeaderIndexMap_(headerRow) {
  var map = {};
  if (!headerRow || !Array.isArray(headerRow)) return map;
  for (var i = 0; i < headerRow.length; i++) {
    var raw = String(headerRow[i] || '').trim().toLowerCase();
    var clean = raw.replace(/[^a-z0-9]/g, '');
    if (clean) map[clean] = i;
  }
  return map;
}

function loginUser(nik, password) {
  try {
    var nikStr = String(nik || '').trim();
    var passStr = String(password || '').trim();
    if (!nikStr || !passStr) return { status: 'error', message: 'NIK dan Password wajib diisi!' };

    var sheetUsers = findSheet_('Users');
    if (!sheetUsers) return { status: 'error', message: 'Sheet Users tidak ditemukan!' };

    var data = sheetUsers.getDataRange().getDisplayValues();
    var matchedUser = null;

    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      if (String(row[0]).trim() === nikStr && String(row[4]).trim() === passStr) {
        matchedUser = {
          nik: row[0],
          phone: row[1],
          email: row[2],
          role: row[3],
          name: row[5],
          point: row[6],
          area: row[7],
          geminiKey: row[9] || '',
          photoUrl: cleanGoogleDriveUrl_(row[10] || ''),
          regional: 'Gorontalo',
          gender: '-',
          masaKerja: '-',
          statusKaryawan: '-',
          posisi: row[3] || '-'
        };
        break;
      }
    }

    if (!matchedUser) {
      return { status: 'error', message: 'NIK atau Password tidak terdaftar!' };
    }

    // Ekstraksi Data MPP Dinamis Berbasis Header
    var sheetMPP = findSheet_('Data MPP');
    if (sheetMPP && sheetMPP.getLastRow() > 1) {
      var mppValues = sheetMPP.getDataRange().getDisplayValues();
      var hMap = findHeaderIndexMap_(mppValues[0]);

      var idxNik = hMap['nik'] !== undefined ? hMap['nik'] : 3;
      var idxReg = hMap['regional'] !== undefined ? hMap['regional'] : (hMap['region'] !== undefined ? hMap['region'] : 0);
      var idxArea = hMap['area'] !== undefined ? hMap['area'] : 1;
      var idxPoint = hMap['point'] !== undefined ? hMap['point'] : (hMap['cabang'] !== undefined ? hMap['cabang'] : 2);
      var idxName = hMap['nama'] !== undefined ? hMap['nama'] : (hMap['namakaryawan'] !== undefined ? hMap['namakaryawan'] : 4);
      var idxGender = hMap['gender'] !== undefined ? hMap['gender'] : (hMap['jeniskelamin'] !== undefined ? hMap['jeniskelamin'] : (hMap['jk'] !== undefined ? hMap['jk'] : 5));
      var idxMasa = hMap['masakerja'] !== undefined ? hMap['masakerja'] : (hMap['masakerjakaryawan'] !== undefined ? hMap['masakerjakaryawan'] : 6);
      var idxStatus = hMap['status'] !== undefined ? hMap['status'] : (hMap['statuskaryawan'] !== undefined ? hMap['statuskaryawan'] : 7);
      var idxPosisi = hMap['posisi'] !== undefined ? hMap['posisi'] : (hMap['jabatan'] !== undefined ? hMap['jabatan'] : (hMap['tipe'] !== undefined ? hMap['tipe'] : 8));

      for (var m = 1; m < mppValues.length; m++) {
        var rNik = String(mppValues[m][idxNik] || '').trim();
        if (rNik === nikStr) {
          matchedUser.regional = mppValues[m][idxReg] || matchedUser.regional;
          matchedUser.area = mppValues[m][idxArea] || matchedUser.area;
          matchedUser.point = mppValues[m][idxPoint] || matchedUser.point;
          matchedUser.name = mppValues[m][idxName] || matchedUser.name;
          matchedUser.gender = mppValues[m][idxGender] || '-';
          matchedUser.masaKerja = mppValues[m][idxMasa] || '-';
          matchedUser.statusKaryawan = mppValues[m][idxStatus] || '-';
          matchedUser.posisi = mppValues[m][idxPosisi] || matchedUser.role;
          break;
        }
      }
    }

    updateHeartbeat(matchedUser);
    return { status: 'success', user: matchedUser };
  } catch (err) {
    return { status: 'error', message: err.toString() };
  }
}

function registerUser(payload) {
  try {
    payload = payload || {};
    var nik = String(payload.nik || '').trim();
    var phone = String(payload.phone || '').trim();
    var email = String(payload.email || '').trim().toLowerCase();
    var role = String(payload.role || '').trim();
    var pass = String(payload.password || '').trim();

    if (!nik || !phone || !email || !role || !pass) {
      return { status: 'error', message: 'Seluruh kolom pendaftaran akun wajib diisi lengkap!' };
    }

    var sheetMPP = findSheet_('Data MPP');
    if (!sheetMPP || sheetMPP.getLastRow() < 2) {
      return { status: 'error', message: 'Sheet Data MPP tidak ditemukan untuk verifikasi NIK!' };
    }

    var mppValues = sheetMPP.getDataRange().getDisplayValues();
    var hMap = findHeaderIndexMap_(mppValues[0]);
    var idxNik = hMap['nik'] !== undefined ? hMap['nik'] : 3;
    var idxPosisi = hMap['posisi'] !== undefined ? hMap['posisi'] : (hMap['jabatan'] !== undefined ? hMap['jabatan'] : 8);
    var idxName = hMap['nama'] !== undefined ? hMap['nama'] : (hMap['namakaryawan'] !== undefined ? hMap['namakaryawan'] : 4);
    var idxPoint = hMap['point'] !== undefined ? hMap['point'] : (hMap['cabang'] !== undefined ? hMap['cabang'] : 2);
    var idxArea = hMap['area'] !== undefined ? hMap['area'] : 1;

    var matchedMpp = null;
    for (var m = 1; m < mppValues.length; m++) {
      var rNik = String(mppValues[m][idxNik] || '').trim();
      if (rNik === nik) {
        matchedMpp = {
          nik: rNik,
          posisi: String(mppValues[m][idxPosisi] || '').trim(),
          name: String(mppValues[m][idxName] || '').trim(),
          point: String(mppValues[m][idxPoint] || '').trim(),
          area: String(mppValues[m][idxArea] || '').trim()
        };
        break;
      }
    }

    if (!matchedMpp) {
      return { 
        status: 'error', 
        message: 'Pendaftaran Ditolak: NIK ' + nik + ' tidak terdaftar dalam database Data MPP Regional Gorontalo!' 
      };
    }

    var cleanRole = role.toLowerCase().replace(/[^a-z0-9]/g, '');
    var cleanPosisi = matchedMpp.posisi.toLowerCase().replace(/[^a-z0-9]/g, '');
    var isMatch = (cleanPosisi === cleanRole) || 
                  (cleanPosisi.indexOf(cleanRole) !== -1) || 
                  (cleanRole.indexOf(cleanPosisi) !== -1);

    if (!isMatch) {
      return {
        status: 'error',
        message: 'Pendaftaran Ditolak: Role ' + role + ' tidak sesuai dengan posisi Anda di Data MPP (' + matchedMpp.posisi + ')!'
      };
    }

    var sheetUsers = findSheet_('Users');
    if (!sheetUsers) return { status: 'error', message: 'Sheet Users tidak ditemukan!' };

    var usersData = sheetUsers.getDataRange().getDisplayValues();
    for (var u = 1; u < usersData.length; u++) {
      if (String(usersData[u][0]).trim() === nik) {
        return { 
          status: 'error', 
          message: 'NIK ' + nik + ' sudah terdaftar sebelumnya! Silakan login atau gunakan menu Lupa Password.' 
        };
      }
    }

    var lock = LockService.getScriptLock();
    try {
      lock.waitLock(5000);
      var regDateStr = Utilities.formatDate(new Date(), CONFIG.TIMEZONE, 'dd/MM/yyyy HH:mm:ss');
      sheetUsers.appendRow([
        nik, phone, email, role, pass,
        matchedMpp.name || ('User ' + nik),
        matchedMpp.point || '-',
        matchedMpp.area || '-',
        regDateStr, '', ''
      ]);
      SpreadsheetApp.flush();
      clearCacheKey_('bp_users_list');
    } finally {
      lock.releaseLock();
    }

    return {
      status: 'success',
      message: 'Pendaftaran Berhasil! Akun personil ' + matchedMpp.name + ' (' + role + ') telah terverifikasi dan aktif.'
    };
  } catch (err) {
    return { status: 'error', message: 'Gagal melakukan pendaftaran akun: ' + err.toString() };
  }
}

function maskEmail_(email) {
  if (!email || email.indexOf('@') === -1) return email || '';
  var parts = email.split('@');
  var name = parts[0];
  var domain = parts[1];
  if (name.length <= 2) {
    return name.charAt(0) + '*@' + domain;
  }
  return name.charAt(0) + '***' + name.charAt(name.length - 1) + '@' + domain;
}

function verifyOtpAndResetPassword(payload) {

    var otpStr = String(payload.otp || '').trim();
    var newPass = String(payload.newPassword || '').trim();

    }

    var cache = CacheService.getScriptCache();
    var cached = cache.get('otp_reset_' + nikStr);
    var isValid = false;

    if (cached) {
      try {
        var obj = JSON.parse(cached);
        if (obj.otp === otpStr && Date.now() <= obj.expires) {
          isValid = true;
        }
      } catch(e) {}
    }

    if (!isValid) {
      var sheetOtp = findSheet_('OTP_Sessions');
      if (sheetOtp && sheetOtp.getLastRow() > 1) {
        var vals = sheetOtp.getDataRange().getDisplayValues();
        for (var i = vals.length - 1; i >= 1; i--) {
          if (String(vals[i][1]).trim() === nikStr && String(vals[i][3]).trim() === otpStr && vals[i][7] === 'Active') {
            isValid = true;
            sheetOtp.getRange(i + 1, 8).setValue('Used');
            break;
          }
        }
      }
    }

    if (!isValid) {
      return { status: 'error', message: 'Kode OTP tidak valid atau sudah kedaluwarsa!' };
    }

    var sheetUsers = findSheet_('Users');

function calculateNameSimilarity_(str1, str2) {
  var s1 = String(str1 || '').toLowerCase().trim().replace(/[^a-z0-9]/g, '');
  var s2 = String(str2 || '').toLowerCase().trim().replace(/[^a-z0-9]/g, '');
  if (!s1 || !s2) return 0;
  if (s1 === s2) return 1.0;

  var longer = s1.length > s2.length ? s1 : s2;
  var shorter = s1.length > s2.length ? s2 : s1;
  var longerLength = longer.length;
  if (longerLength === 0) return 1.0;

  var costs = [];
  for (var i = 0; i <= longer.length; i++) {
    var lastValue = i;
    for (var j = 0; j <= shorter.length; j++) {
      if (i === 0) costs[j] = j;
      else {
        if (j > 0) {
          var newValue = costs[j - 1];
          if (longer.charAt(i - 1) !== shorter.charAt(j - 1)) {
            newValue = Math.min(Math.min(newValue, lastValue), costs[j]) + 1;
          }
          costs[j - 1] = lastValue;
          lastValue = newValue;
        }
      }
    }
    if (i > 0) costs[shorter.length] = lastValue;
  }

  return (longerLength - costs[shorter.length]) / parseFloat(longerLength);
}

    var role = String(userSession.role).trim();
    if (role !== '1-RM' && role !== '2-AM' && role !== 'Editor' && role !== 'Admin') {
      return { status: 'error', message: 'Hak akses terbatas! Hanya Role 1-RM dan 2-AM yang berhak mengunggah berkas Bahan Kunjungan.' };
    }

    payload = payload || {};
    var menuType = String(payload.menuType || '').trim(); // 'Renewal' atau 'Reactivation'
    var fileName = String(payload.fileName || 'Data_Unggahan.xlsx').trim();
    var rowsData = payload.rowsData || [];
    var customHeaders = payload.headers || [];

    if (menuType !== 'Renewal' && menuType !== 'Reactivation') {
      return { status: 'error', message: 'Tipe menu bahan kunjungan tidak valid!' };
    }

    if (!rowsData || rowsData.length === 0) {
      return { status: 'error', message: 'Data berkas kosong atau tidak ada baris yang valid untuk diproses!' };
    }

    var targetSheetName = (menuType === 'Renewal') ? 'Renewal_Data' : 'Reactivation_Data';
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) return { status: 'error', message: 'Spreadsheet tidak aktif!' };

    var sheet = ss.getSheetByName(targetSheetName);
    if (!sheet) {
      sheet = ss.insertSheet(targetSheetName);
    }

    var nowStr = Utilities.formatDate(new Date(), CONFIG.TIMEZONE, 'dd/MM/yyyy HH:mm:ss') + ' WITA';
    var uploaderRole = userSession.role || 'User';
    var uploaderName = userSession.name || userSession.nik;
    var uploaderInfo = uploaderRole + ' ' + uploaderName;

    // Ekstraksi header dinamis dari file impor
    var headers = [];
    if (customHeaders && customHeaders.length > 0) {
      headers = customHeaders.slice();
    } else if (rowsData.length > 0) {
      headers = Object.keys(rowsData[0]);
    }

    headers = headers.filter(function(h) {
      return h && String(h).indexOf('_raw_') === -1;
    });

    var lock = LockService.getScriptLock();
    try {
      lock.waitLock(10000);

      sheet.clearContents();
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
      sheet.getRange(1, 1, 1, headers.length)
           .setFontWeight('bold')
           .setBackground('#101828')
           .setFontColor('#ffffff');

      var matrix = [];
      for (var r = 0; r < rowsData.length; r++) {
        var rowObj = rowsData[r];
        var rowArr = [];
        for (var c = 0; c < headers.length; c++) {
          var colKey = headers[c];
          var cellVal = rowObj[colKey];
          if (cellVal === undefined || cellVal === null) cellVal = '';
          rowArr.push(cellVal);
        }
        matrix.push(rowArr);
      }

      if (matrix.length > 0) {
        var batchSize = 500;
        for (var b = 0; b < matrix.length; b += batchSize) {
          var chunk = matrix.slice(b, b + batchSize);
          sheet.getRange(b + 2, 1, chunk.length, headers.length).setValues(chunk);
        }
      }

      var auditSheet = ss.getSheetByName('Bahan_Kunjungan_Audit');
      if (!auditSheet) {
        auditSheet = ss.insertSheet('Bahan_Kunjungan_Audit');
        auditSheet.appendRow(['Audit_ID', 'Timestamp_WITA', 'Menu_Tipe', 'Nama_File', 'Total_Baris', 'Uploaded_By_NIK', 'Uploaded_By_Name', 'Uploaded_By_Role']);

        'AUD-' + Date.now(),
        nowStr,
        menuType,
        fileName,
        rowsData.length,
        userSession.nik || '',
        uploaderName,
        uploaderRole
      ]);

      SpreadsheetApp.flush();
      clearSheetCache_((menuType === 'Renewal') ? 'renewal' : 'reactivation');
    } finally {
      lock.releaseLock();
    }

    return {
      status: 'success',
      message: 'Berkas ' + fileName + ' (' + rowsData.length + ' baris) berhasil diunggah dengan ' + headers.length + ' kolom dinamis untuk Menu ' + menuType + '!',
      audit: {
        uploaderRole: uploaderRole,
        uploaderName: uploaderName,
        timestampWITA: nowStr,
        fileName: fileName,
        totalRows: rowsData.length
      }
    };

  }

function verifyPostDisb(payload, userSession) {
  try {
    if (!userSession || !userSession.role) {
      return { status: 'error', message: 'Akses ditolak! Silakan login.' };
    }

    var role = String(userSession.role || '').trim();
    if (role !== '1-RM' && role !== '2-AM' && role !== '3-BM' && role !== 'Editor' && role !== 'Admin') {
      return { status: 'error', message: 'Hanya Role 1-RM, 2-AM, dan 3-BM yang berhak memverifikasi Post Disb!' };
    }

    payload = payload || {};
    var loanId = String(payload.loan_id || payload.idloan || '').trim();
    var custNum = String(payload.customer_number || payload.customernumber || '').trim();
    var custName = String(payload.customer_name || payload.customername || '').trim();
    var statusChoice = String(payload.status_choice || 'Sudah Post Disb').trim();

    if (!loanId && !custNum) {
      return { status: 'error', message: 'Nomor Loan / Customer Number tidak valid!' };
    }

    var proofBase64 = payload.proofBase64 || '';
    var proofMimeType = String(payload.proofMimeType || '').toLowerCase().trim();
    var proofFileName = payload.proofFileName || ('Bukti_' + (loanId || custNum));
    var proofUrl = '-';

    if (statusChoice === 'Sudah Post Disb' || statusChoice === 'Tidak Direspon') {
      if (statusChoice === 'Sudah Post Disb' && role === '3-BM') {
        if (!proofBase64 || proofMimeType.indexOf('video') === -1) {
          return { status: 'error', message: 'Verifikasi Ditolak: Khusus Role 3-BM WAJIB melampirkan bukti format VIDEO (.mp4/.webm/.mov/.3gp)!' };
        }
      } else {
        if (proofBase64) {
          var isAllowed = (proofMimeType.indexOf('image') !== -1 || proofMimeType.indexOf('video') !== -1 || proofMimeType.indexOf('audio') !== -1);
          if (!isAllowed) {
            return { status: 'error', message: 'Format bukti hanya mendukung Gambar (PNG/JPG), Video, atau Rekaman Suara!' };
          }
        }
      }

      if (proofBase64 && proofBase64.indexOf('base64,') !== -1) {
        try {
          var folderName = 'Bukti_Post_Disb_Regional_Gorontalo';
          var folders = DriveApp.getFoldersByName(folderName);
          var targetFolder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
          targetFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

          var cleanData = proofBase64.split('base64,')[1];
          var decodedBytes = Utilities.base64Decode(cleanData);
          var ext = '.jpg';
          var isVideo = (proofMimeType.indexOf('video') !== -1);
          var isAudio = (proofMimeType.indexOf('audio') !== -1);
          var isPng = (proofMimeType.indexOf('png') !== -1);

          if (isVideo) ext = '.mp4';
          else if (isAudio) ext = '.mp3';
          else if (isPng) ext = '.png';

          var blob = Utilities.newBlob(decodedBytes, proofMimeType, proofFileName.replace(/\.[^/.]+$/, "") + '_' + Date.now() + ext);
          var file = targetFolder.createFile(blob);
          file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
          
          var fileId = file.getId();
          proofUrl = (isVideo || isAudio) ? ('https://drive.google.com/file/d/' + fileId + '/preview') : ('https://lh3.googleusercontent.com/d/' + fileId);
        } catch (eDrive) {
          Logger.log('Error upload bukti Post Disb: ' + eDrive.toString());
        }
      }
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) return { status: 'error', message: 'Spreadsheet tidak aktif!' };

    var sheet = ss.getSheetByName('Verifikasi_PostDisb');
    if (!sheet) {
      sheet = ss.insertSheet('Verifikasi_PostDisb');
      sheet.appendRow([
        'ID', 'Loan ID', 'Customer Number', 'Customer Name', 'Plafond',
        'Disbursement Date', 'Tanggal Approve UK', 'BP', 'Point', 'Area',
        'Verified By Role', 'Verified By Name', 'Verified At', 'Phone No', 'Status Post Disb', 'Bukti URL', 'Catatan'
      ]);
    }

    var lock = LockService.getScriptLock();
    try {
      lock.waitLock(5000);
      var nowStr = Utilities.formatDate(new Date(), CONFIG.TIMEZONE, 'dd/MM/yyyy HH:mm:ss');
      var verId = 'VPD-' + Date.now();

      sheet.appendRow([
        verId, loanId, custNum, custName,
        payload.plafond || '',
        payload.disbursement_date || '',
        payload.tanggal_approve_uk || '',
        payload.bp || '',
        payload.branch || payload.point || '',
        payload.area || '',
        userSession.role || '',
        userSession.name || '',
        nowStr,
        payload.phone_no || '',
        statusChoice,
        proofUrl,
        payload.notes || '-'
      ]);

      SpreadsheetApp.flush();
      clearSheetCache_('approvalPerhari');
      clearSheetCache_('postDisbLog');
    } finally {
      lock.releaseLock();
    }

    return {
      status: 'success',
      proofUrl: proofUrl,
      verifiedRole: userSession.role || '',
      verifiedName: userSession.name || '',
      statusChoice: statusChoice,
      message: 'Status mitra ' + (custName || custNum) + ' berhasil diverifikasi sebagai: ' + statusChoice
    };
  } catch (err) {
    return { status: 'error', message: err.toString() };
  }
}

function getPerformaKpPerWeekSummary(userSession, filterArea, filterPoint, filterBP) {
  try {
    if (!userSession || !userSession.role) {
      return { status: 'error', message: 'Akses ditolak! Silakan login.' };
    }

    var role = String(userSession.role).trim();
    var userArea = String(userSession.area || '').trim().toLowerCase();
    var userPoint = String(userSession.point || '').trim().toLowerCase();
    var userNik = String(userSession.nik || '').trim();
    var isBpRole = (role === '4-BP' || role === '5-BPR' || role === '6-BP Buffer');

    var appResult = getRawDataWithHeaders_('approvalPerhari');
    var rows = appResult.rows || [];
    if (rows.length === 0) {
      return { status: 'success', areaSummary: [], pointSummary: [], bpSummary: [] };
    }

    var fArea = String(filterArea || '').trim().toLowerCase();
    var fPoint = String(filterPoint || '').trim().toLowerCase();
    var fBP = String(filterBP || '').trim().toLowerCase();

    var dayNames = ['Senin', 'Selasa', 'Rabu', 'Kamis', 'Jumat', 'Sabtu'];
    var weeksKey = ['W1', 'W2', 'W3', 'W4', 'W5'];

    var areaMap = {}, pointMap = {}, bpMap = {};

    rows.forEach(function(r) {
      var rArea = String(r.area || r['_raw_1'] || '').trim();
      var rPoint = String(r.point || r.branch || r['_raw_2'] || '').trim();
      var rNik = String(r.nik || r.nik_bp || r['_raw_12'] || '').trim();
      var rName = String(r.bp || r.bpbm || r['_raw_13'] || '').trim();

      if (role === '2-AM' && userArea && rArea.toLowerCase() !== userArea) return;
      if (role === '3-BM' && userPoint && rPoint.toLowerCase() !== userPoint) return;
      if (isBpRole && userNik && rNik !== userNik) return;

      if (fArea && rArea.toLowerCase() !== fArea) return;
      if (fPoint && rPoint.toLowerCase() !== fPoint) return;
      if (fBP && (rName.toLowerCase().indexOf(fBP) === -1 && rNik.indexOf(fBP) === -1)) return;

      var rDateStr = r.disbursement_date || r.tanggal_approve_uk || r.tanggal || r['_raw_8'] || '';
      var dObj = parseDateAny_(rDateStr);
      if (!dObj || isNaN(dObj.getTime())) return;
      var dDay = dObj.getDay();
      if (dDay === 0) return;

      var dayIdx = dDay - 1;
      var dayName = dayNames[dayIdx];

      var dayOfMonth = dObj.getDate();
      var weekIdx = Math.min(4, Math.floor((dayOfMonth - 1) / 7));
      var weekLabel = weeksKey[weekIdx];

      var plf = parseFloat(String(r.plafond || r['_raw_7'] || '0').replace(/[^0-9]/g, '')) || 0;

      if (!areaMap[rArea]) areaMap[rArea] = initEntityPerWeek_(rArea, 'Area');
      accumulateEntityAmount_(areaMap[rArea], weekLabel, dayName, plf);

      var ptKey = rArea + '_' + rPoint;
      if (!pointMap[ptKey]) pointMap[ptKey] = initEntityPerWeek_(rPoint, 'Point', rArea);
      accumulateEntityAmount_(pointMap[ptKey], weekLabel, dayName, plf);

      var bpKey = rNik || rName;
      if (!bpMap[bpKey]) bpMap[bpKey] = initEntityPerWeek_(rName, 'BP', rArea, rPoint, rNik);
      accumulateEntityAmount_(bpMap[bpKey], weekLabel, dayName, plf);
    });

    var areaList = finalizeWeekComparison_(areaMap);
    var pointList = finalizeWeekComparison_(pointMap);
    var bpList = finalizeWeekComparison_(bpMap);

    return {
      status: 'success',
      areaSummary: areaList,
      pointSummary: pointList,
      bpSummary: bpList
    };
  } catch (err) {
    return { status: 'error', message: err.toString() };
  }
}

function initEntityPerWeek_(name, type, area, point, nik) {
  return {
    name: name,
    type: type,
    area: area || name,
    point: point || '-',
    nik: nik || '-',
    weeks: {
      W1: { Senin:0, Selasa:0, Rabu:0, Kamis:0, Jumat:0, Sabtu:0, total:0 },
      W2: { Senin:0, Selasa:0, Rabu:0, Kamis:0, Jumat:0, Sabtu:0, total:0 },
      W3: { Senin:0, Selasa:0, Rabu:0, Kamis:0, Jumat:0, Sabtu:0, total:0 },
      W4: { Senin:0, Selasa:0, Rabu:0, Kamis:0, Jumat:0, Sabtu:0, total:0 },
      W5: { Senin:0, Selasa:0, Rabu:0, Kamis:0, Jumat:0, Sabtu:0, total:0 }
    },
    totalBulan: 0
  };
}

function accumulateEntityAmount_(entity, week, day, amount) {
  if (entity.weeks[week] && entity.weeks[week][day] !== undefined) {
    entity.weeks[week][day] += amount;
    entity.weeks[week].total += amount;
    entity.totalBulan += amount;
  }
}

function finalizeWeekComparison_(map) {
  var list = [];
  var days = ['Senin', 'Selasa', 'Rabu', 'Kamis', 'Jumat', 'Sabtu'];

  Object.keys(map).forEach(function(k) {
    var item = map[k];
    item.comparisons = [];

    var pairs = [
      { from: 'W2', to: 'W1', label: 'W2 vs W1' },
      { from: 'W3', to: 'W2', label: 'W3 vs W2' },
      { from: 'W4', to: 'W3', label: 'W4 vs W3' }
    ];

    pairs.forEach(function(p) {
      var wCurr = item.weeks[p.from];
      var wPrev = item.weeks[p.to];
      var dayDeltas = {};

      days.forEach(function(d) {
        var diff = wCurr[d] - wPrev[d];
        var pct = wPrev[d] > 0 ? ((diff / wPrev[d]) * 100).toFixed(1) : (diff > 0 ? '100.0' : '0.0');
        dayDeltas[d] = {
          diff: diff,
          pct: pct,
          status: diff >= 0 ? 'naik' : 'turun'
        };
      });

      var totalDiff = wCurr.total - wPrev.total;
      var totalPct = wPrev.total > 0 ? ((totalDiff / wPrev.total) * 100).toFixed(1) : (totalDiff > 0 ? '100.0' : '0.0');

      item.comparisons.push({
        pairLabel: p.label,
        currWeek: p.from,
        prevWeek: p.to,
        dayDeltas: dayDeltas,
        totalDiff: totalDiff,
        totalPct: totalPct,
        status: totalDiff >= 0 ? 'naik' : 'turun'
      });
    });

    var activeWeeks = 0;
    if (item.weeks.W1.total > 0) activeWeeks++;
    if (item.weeks.W2.total > 0) activeWeeks++;
    if (item.weeks.W3.total > 0) activeWeeks++;
    if (item.weeks.W4.total > 0) activeWeeks++;

    var weeklyRunRate = activeWeeks > 0 ? Math.round(item.totalBulan / activeWeeks) : 0;
    item.weeklyRunRate = weeklyRunRate;
    item.projectionEOM = weeklyRunRate * 4;

    list.push(item);
  });

  list.sort(function(a, b) { return b.totalBulan - a.totalBulan; });
  return list;
}

function getRawDataWithHeaders_(sheetKey) {
  var cache = CacheService.getScriptCache();
  var cacheKey = 'raw_sheet_v15_' + sheetKey;

  var cachedCount = cache.get(cacheKey + '_count');
  if (cachedCount) {
    try {
      var count = parseInt(cachedCount, 10);
      var jsonStr = '';
      for (var i = 0; i < count; i++) {
        var chunk = cache.get(cacheKey + '_' + i);
        if (chunk === null) { jsonStr = ''; break; }
        jsonStr += chunk;
      }
      if (jsonStr) {
        return JSON.parse(jsonStr);
      }
    } catch (e) {}
  }

  var cfg = CONFIG.SHEETS[sheetKey];
  var sheet = cfg ? findSheet_(cfg.name) : findSheet_(sheetKey);
  if (!sheet) return { rows: [], headers: [] };
  
  var result = readSheetWithHeaders_(sheet);

  try {
    var jsonToCache = JSON.stringify(result);
    var chunkSize = 90000;
    var numChunks = Math.ceil(jsonToCache.length / chunkSize);
    var cacheObj = {};
    cacheObj[cacheKey + '_count'] = String(numChunks);
    for (var c = 0; c < numChunks; c++) {
      cacheObj[cacheKey + '_' + c] = jsonToCache.substring(c * chunkSize, (c + 1) * chunkSize);
    }
    cache.putAll(cacheObj, CONFIG.CACHE_TTL);
  } catch (e) {}

  return result;
}

function readSheetWithHeaders_(sheet) {
  if (!sheet) return { rows: [], headers: [] };
  var lastRow = sheet.getLastRow();
  var lastColumn = sheet.getLastColumn();
  if (lastRow < 2 || lastColumn < 1) return { rows: [], headers: [] };

  var sName = sheet.getName().toLowerCase();
  var isLoanPaid = (sName.indexOf('loan paid') !== -1);

  if (isLoanPaid) {
    var values = sheet.getRange(1, 1, lastRow, lastColumn).getDisplayValues();
    var isRegional = (sName.indexOf('regional') !== -1);
    var startRowIdx = isRegional ? 2 : 3;
    var headers = [];

    var currentSection = '';
    for (var c = 0; c < lastColumn; c++) {
      var s0 = String(values[0][c] || '').trim();
      if (s0) currentSection = s0;

      if (isRegional) {
        var m1 = String(values[1][c] || '').trim();
        var label = m1 || values[0][c] || ('Col' + (c + 1));
        if (m1 && currentSection && m1.toLowerCase() !== currentSection.toLowerCase()) {
          label = currentSection + ' ' + m1;
        }
        headers.push(label);
      } else {
        var t1 = String(values[1][c] || '').trim();
        var m2 = String(values[2][c] || '').trim();
        var label = m2 || t1 || currentSection || ('Col' + (c + 1));
        if (m2 && currentSection && m2.toLowerCase() !== currentSection.toLowerCase()) {
          label = currentSection + ' ' + m2 + (t1 ? ' (' + t1 + ')' : '');
        } else if (m2 && t1) {
          label = m2 + ' (' + t1 + ')';
        }
        headers.push(label);
      }
    }

    var rows = [];
    for (var r = startRowIdx; r < values.length; r++) {
      var row = {};
      var hasValue = false;
      for (var col = 0; col < lastColumn; col++) {
        var hName = headers[col];
        var key = canonicalHeader_(hName);
        var val = values[r][col] === undefined ? '' : values[r][col];
        if (String(val).trim() !== '') hasValue = true;
        row[key] = val;
        row['_raw_' + col] = val;
      }
      if (hasValue) rows.push(row);
    }
    return { rows: rows, headers: headers };
  }

  // Pembacaan Dinamis Sheet Impor Bahan Kunjungan (Renewal & Reactivation)
  if (sName === 'renewal_data' || sName === 'reactivation_data') {
    var uValues = sheet.getRange(1, 1, lastRow, lastColumn).getDisplayValues();
    var uHeaders = uValues[0];
    var uRows = [];

    for (var ur = 1; ur < uValues.length; ur++) {
      var uRow = {};
      var uHasVal = false;
      for (var uc = 0; uc < uHeaders.length; uc++) {
        var uHName = String(uHeaders[uc] || ('Col' + (uc + 1))).trim();
        var uKey = canonicalHeader_(uHName);
        var uVal = uValues[ur][uc] === undefined ? '' : uValues[ur][uc];
        if (String(uVal).trim() !== '') uHasVal = true;
        uRow[uHName] = uVal;
        if (!uRow[uKey]) uRow[uKey] = uVal;
        uRow['_raw_' + uc] = uVal;
      }
      if (uHasVal) uRows.push(uRow);
    }
    return { rows: uRows, headers: uHeaders };
  }

  var values = sheet.getRange(1, 1, lastRow, lastColumn).getDisplayValues();
  var headers = values[0];
  var rows = [];

  for (var r = 1; r < values.length; r++) {
    var row = {};
    var hasValue = false;
    for (var c = 0; c < headers.length; c++) {
      var header = String(headers[c] || ('Col' + (c + 1))).trim();
      var key = canonicalHeader_(header);
      var val = values[r][c] === undefined ? '' : values[r][c];
      if (String(val).trim() !== '') hasValue = true;
      row[key] = val;
      row['_raw_' + c] = val;
    }
    if (hasValue) rows.push(row);
  }
  return { rows: rows, headers: headers };
}

function findSheet_(sheetName) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) return null;
    var target = String(sheetName || '').trim().toLowerCase();
    if (target === 'db_sos' || target === 'dbsos') return null;

    var exact = ss.getSheetByName(sheetName);
    if (exact && exact.getName().trim().toLowerCase() !== 'db_sos') return exact;

    var sheets = ss.getSheets();
    for (var i = 0; i < sheets.length; i++) {
      var sName = sheets[i].getName().trim().toLowerCase();
      if (sName !== 'db_sos' && sName === target) return sheets[i];
    }
    return null;
  } catch (err) { return null; }
}

function updateHeartbeat(userSession, lat, lng) {
  try {
    if (!userSession || !userSession.nik) return { status: 'error' };
    var nik = String(userSession.nik).trim();
    var now = new Date();
    var timeStr = Utilities.formatDate(now, CONFIG.TIMEZONE, 'dd/MM/yyyy HH:mm:ss');
    var millis = now.getTime();

    var cache = CacheService.getScriptCache();
    var existingRaw = cache.get('active_user_' + nik);
    var history = [];
    var lastLat = parseFloat(lat) || null;
    var lastLng = parseFloat(lng) || null;
    var existingPhoto = cleanGoogleDriveUrl_(userSession.photoUrl || '');

    if (existingRaw) {
      try {
        var existingData = JSON.parse(existingRaw);
        history = existingData.history || [];
        if (!lastLat && !lastLng) {
           lastLat = existingData.lastLat || null;
           lastLng = existingData.lastLng || null;
        }
        if (!existingPhoto && existingData.photoUrl) {
          existingPhoto = cleanGoogleDriveUrl_(existingData.photoUrl);
        }
      } catch(e) {}
    }
    
    // Merekam jejak rute (breadcrumb trail) selama user aktif
    if (lastLat && lastLng) {
      var isSameLocation = false;
      if (history.length > 0) {
         var lastHist = history[history.length - 1];
         if (lastHist.lat === lastLat && lastHist.lng === lastLng) isSameLocation = true;
      }
      if (!isSameLocation) {
         history.push({ lat: lastLat, lng: lastLng, time: timeStr, millis: millis });
         if (history.length > 150) history.shift();
      }
    }

    var userObj = {
      nik: nik, 
      name: userSession.name || '', 
      role: userSession.role || '',
      point: userSession.point || '', 
      area: userSession.area || '',
      lastActive: timeStr, 
      timestampMillis: millis, 
      isOnline: true,
      lastLat: lastLat, 
      lastLng: lastLng,
      photoUrl: existingPhoto,
      history: history
    };
    
    cache.put('active_user_' + nik, JSON.stringify(userObj), 86400);

    var activeNiksRaw = cache.get('active_niks_list') || '[]';
    var activeNiks = [];
    try { activeNiks = JSON.parse(activeNiksRaw); } catch(e) { activeNiks = []; }
    if (activeNiks.indexOf(nik) === -1) {
      activeNiks.push(nik);
      cache.put('active_niks_list', JSON.stringify(activeNiks), 86400);
    }
    return { status: 'success' };
  } catch (err) { return { status: 'error', message: err.toString() }; }
}

function getOnlineUsers() {
  try {
    var cache = CacheService.getScriptCache();
    var activeNiksRaw = cache.get('active_niks_list') || '[]';
    var activeNiks = [];
    try { activeNiks = JSON.parse(activeNiksRaw); } catch(e) { activeNiks = []; }

    var nowMillis = new Date().getTime();
    var activeThreshold = 3 * 60 * 1000;
    var onlineUsers = [];
    var validNiks = [];

    activeNiks.forEach(function(nik) {
      var userRaw = cache.get('active_user_' + nik);
      if (userRaw) {
        try {
          var u = JSON.parse(userRaw);
          var isOnline = (nowMillis - (u.timestampMillis || 0)) <= activeThreshold;
          if (isOnline) {
            u.isOnline = isOnline;
            u.photoUrl = cleanGoogleDriveUrl_(u.photoUrl);
            onlineUsers.push(u);
          }
          validNiks.push(nik);
        } catch(e) {}
      }
    });

    if (validNiks.length !== activeNiks.length) {
      cache.put('active_niks_list', JSON.stringify(validNiks), 86400);
    }
    return onlineUsers;
  } catch (err) { return []; }
}

function getTrackingData(userSession) {
  try {
    if (!userSession || !userSession.role) return [];
    var role = String(userSession.role).trim();
    var userArea = String(userSession.area || '').trim().toLowerCase();
    var userPoint = String(userSession.point || '').trim().toLowerCase();
    
    if (role !== '1-RM' && role !== '2-AM' && role !== '3-BM' && role !== 'Editor' && role !== 'Admin') return [];

    var cache = CacheService.getScriptCache();
    var activeNiksRaw = cache.get('active_niks_list') || '[]';
    var activeNiks = [];
    try { activeNiks = JSON.parse(activeNiksRaw); } catch(e) {}

    var nowMillis = new Date().getTime();
    var trackingUsers = [];
    var oneDayMillis = 24 * 60 * 60 * 1000;

    activeNiks.forEach(function(nik) {
      var userRaw = cache.get('active_user_' + nik);
      if (userRaw) {
        try {
          var u = JSON.parse(userRaw);
          var uRole = String(u.role).trim();
          var uArea = String(u.area).trim().toLowerCase();
          var uPoint = String(u.point).trim().toLowerCase();
          
          var allowed = false;
          if (role === '1-RM' || role === 'Editor' || role === 'Admin') allowed = true;
          else if (role === '2-AM' && (uArea === userArea)) allowed = true;
          else if (role === '3-BM' && (uPoint === userPoint)) allowed = true;

          var isBpRole = (uRole === '4-BP' || uRole === '5-BPR' || uRole === '6-BP Buffer');
          var timeDiff = nowMillis - (u.timestampMillis || 0);

          if (allowed && isBpRole && u.lastLat && u.lastLng && (timeDiff <= oneDayMillis)) {
            var diffMins = Math.floor(timeDiff / 60000);
            var statusColor = '#22c55e';
            if (diffMins >= 5 && diffMins <= 30) statusColor = '#eab308';
            else if (diffMins > 30) statusColor = '#ef4444';

            u.statusColor = statusColor;
            u.diffMins = diffMins;
            u.photoUrl = cleanGoogleDriveUrl_(u.photoUrl);

            var totalDistance = 0;
            if (u.history && u.history.length > 1) {
              for (var h = 0; h < u.history.length - 1; h++) {
                totalDistance += calculateDistanceMeters_(u.history[h].lat, u.history[h].lng, u.history[h+1].lat, u.history[h+1].lng);
              }
            }
            u.totalDistanceKm = (totalDistance / 1000).toFixed(2);
            trackingUsers.push(u);
          }
        } catch(e) {}
      }
    });

    return trackingUsers;
  } catch (err) { return []; }
}

function getGlobalChatMessages(userSession) {
  try {
    var rawResult = getRawDataWithHeaders_('GlobalChat');
    var values = rawResult.rows || [];
    if (values.length === 0) return [];

    var userNik = userSession ? String(userSession.nik || '').trim() : '';
    var userName = userSession ? String(userSession.name || '').trim().toLowerCase() : '';
    var userPoint = userSession ? String(userSession.point || '').trim().toLowerCase() : '';
    var userArea = userSession ? String(userSession.area || '').trim().toLowerCase() : '';

    var msgs = [];
    values.forEach(function(r) {
      var readByRaw = String(r['_raw_9'] || r.readby || '');
      var readByList = readByRaw.split(',').map(function(s) { return s.trim(); }).filter(Boolean);
      var senderNikStr = String(r['_raw_2'] || r.sendernik || '').trim();

      var isRead = false;
      if (userNik) {
        if (senderNikStr === userNik) {
          var readersOtherThanSender = readByList.filter(function(nik) { return nik !== userNik; });
          isRead = readersOtherThanSender.length > 0;
        } else {
          isRead = readByList.indexOf(userNik) !== -1;
        }
      }

      var msg = {
        timestamp: r['_raw_0'] || r.timestamp || '',
        senderName: r['_raw_1'] || r.sendername || '',
        senderNik: senderNikStr,
        message: r['_raw_3'] || r.message || '',
        scope: r['_raw_4'] || r.scope || 'all',
        targetValue: r['_raw_5'] || r.targetvalue || '',
        senderRole: r['_raw_6'] || r.senderrole || '',
        senderPoint: r['_raw_7'] || r.senderpoint || '',
        senderArea: r['_raw_8'] || r.senderarea || '',
        readBy: readByRaw,
        isRead: isRead
      };

      if (userNik && msg.senderNik === userNik) {
        msgs.push(msg);
        return;
      }

      if (msg.scope === 'all') {
        msgs.push(msg);
        return;
      }

      if (msg.scope === 'area' && userArea && String(msg.senderArea).toLowerCase() === userArea) {
        msgs.push(msg);
        return;
      }

      if (msg.scope === 'point' && userPoint && String(msg.senderPoint).toLowerCase() === userPoint) {
        msgs.push(msg);
        return;
      }

      if (msg.scope === 'direct') {
        var targetStr = String(msg.targetValue || '').trim().toLowerCase();
        if (targetStr === userNik.toLowerCase() || targetStr === userName) {
          msgs.push(msg);
        }
      }
    });
    return msgs;
  } catch (err) { return []; }
}

function markChatMessagesAsRead(userSession) {
  try {
    if (!userSession || !userSession.nik) return { status: 'error' };
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) return { status: 'error' };
    var sheet = ss.getSheetByName('GlobalChat');
    if (!sheet || sheet.getLastRow() < 2) return { status: 'success' };

    var lock = LockService.getScriptLock();
    try {
      lock.waitLock(5000);
      var userNik = String(userSession.nik).trim();
      var numRows = sheet.getLastRow() - 1;
      var range = sheet.getRange(2, 10, numRows, 1);
      var values = range.getDisplayValues();
      var hasChanges = false;

      for (var i = 0; i < values.length; i++) {
        var currentRead = String(values[i][0] || '');
        if (currentRead.indexOf(userNik) === -1) {
          values[i][0] = currentRead ? (currentRead + ',' + userNik) : userNik;
          hasChanges = true;
        }
      }

      if (hasChanges) {
        range.setValues(values);
        SpreadsheetApp.flush();
        clearSheetCache_('GlobalChat');
      }
    } finally {
      lock.releaseLock();
    }
    return { status: 'success' };
  } catch (err) { return { status: 'error', message: err.toString() }; }
}

function calculateDistanceMeters_(lat1, lon1, lat2, lon2) {
  var R = 6371000;
  var dLat = (lat2 - lat1) * Math.PI / 180;
  var dLon = (lon2 - lon1) * Math.PI / 180;
  var a = Math.sin(dLat / 2) * Math.sin(dLat / 2) + Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) * Math.sin(dLon / 2) * Math.sin(dLon / 2);
  var c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
  return R * c;
}

function submitAbsensi(payload, userSession) {
  try {
    if (!userSession || !userSession.nik) {
      return { status: 'error', message: 'Akses ditolak! Silakan login untuk melakukan absensi.' };
    }

    payload = payload || {};
    var type = payload.type || 'Individu';
    var userLat = parseFloat(payload.latitude) || 0;
    var userLng = parseFloat(payload.longitude) || 0;
    var selectedPoint = String(payload.point || userSession.point || '').trim();
    var photoBase64 = payload.photoBase64 || '';
    var remarks = String(payload.keterangan || '').trim();
    var inputStatus = String(payload.statusAbsen || 'Hadir').trim();
    var isIzinSakit = (inputStatus === 'Izin' || inputStatus === 'Sakit');

    if (!selectedPoint || selectedPoint === '-' || selectedPoint === 'Point Tidak Terpilih' || selectedPoint === 'Point') {
      return { status: 'error', message: 'Validasi Gagal: Anda WAJIB memilih Point Master penempatan lokasi absensi Anda terlebih dahulu!' };
    }

    var now = new Date();
    var dateStr = Utilities.formatDate(now, CONFIG.TIMEZONE, 'dd/MM/yyyy');
    var timeStr = Utilities.formatDate(now, CONFIG.TIMEZONE, 'HH:mm:ss');
    var hour = parseInt(Utilities.formatDate(now, CONFIG.TIMEZONE, 'HH'), 10);
    var minute = parseInt(Utilities.formatDate(now, CONFIG.TIMEZONE, 'mm'), 10);
    var totalMinutes = hour * 60 + minute;

    var maxAllowedMinutes = isIzinSakit ? (15 * 60 + 30) : (9 * 60);
    if (totalMinutes < (6 * 60) || totalMinutes > maxAllowedMinutes) {
      var limitText = isIzinSakit ? 'antara pukul 06:00 hingga 15:30 WITA' : 'antara pukul 06:00 hingga 09:00 WITA';
      return { status: 'error', message: 'Sistem menolak! Absensi berstatus ' + inputStatus + ' hanya dapat dilakukan ' + limitText + '.' };
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) return { status: 'error', message: 'Spreadsheet tidak aktif!' };

    var absensiSheet = ss.getSheetByName('Absensi');
    if (!absensiSheet) return { status: 'error', message: 'Sheet Absensi belum dikonfigurasi!' };

    var absensiData = absensiSheet.getDataRange().getDisplayValues();
    var currentUserNik = String(userSession.nik).trim();

    if (type === 'Individu') {
      for (var r = 1; r < absensiData.length; r++) {
        var rTanggal = String(absensiData[r][1] || '').trim();
        var rNik = String(absensiData[r][3] || '').trim();
        if (rNik === currentUserNik && isSameDay_(rTanggal, now)) {
          return { status: 'error', message: 'Anda sudah absen hari ini.' };
        }
      }
    }

    var masterPoints = getMasterPoints();
    var targetPointData = null;
    for (var p = 0; p < masterPoints.length; p++) {
      if (String(masterPoints[p].point).trim().toLowerCase() === selectedPoint.toLowerCase()) {
        targetPointData = masterPoints[p];
        break;
      }
    }

    var distanceMeters = 0;
    if (targetPointData && targetPointData.latitude && targetPointData.longitude) {
      distanceMeters = Math.round(calculateDistanceMeters_(userLat, userLng, targetPointData.latitude, targetPointData.longitude));
      if (!isIzinSakit && (inputStatus === 'Hadir' || inputStatus === 'Terlambat') && distanceMeters > targetPointData.radiusMeter) {
        return {
          status: 'error',
          message: 'Absensi Gagal! Anda berada di luar radius point ' + targetPointData.point + ' (' + distanceMeters + 'm dari max ' + targetPointData.radiusMeter + 'm).'
        };
      }
    }

    var photoUrl = '-';
    if (photoBase64 && photoBase64.indexOf('base64,') !== -1) {
      try {
        var folderName = 'Dokumentasi Absensi Regional Gorontalo';
        var folders = DriveApp.getFoldersByName(folderName);
        var targetFolder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
        targetFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

        var base64Data = photoBase64.split('base64,')[1];
        var decodedBlob = Utilities.newBlob(Utilities.base64Decode(base64Data), 'image/jpeg', 'Absen_' + userSession.nik + '_' + Date.now() + '.jpg');
        var file = targetFolder.createFile(decodedBlob);
        file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        photoUrl = 'https://lh3.googleusercontent.com/d/' + file.getId();
      } catch (eDrive) {
        Logger.log('Error simpan foto Drive: ' + eDrive.toString());
      }
    }

    var createdBy = userSession.name + ' (' + userSession.nik + ')';
    var absensiRecords = [];
    var cache = CacheService.getScriptCache();

    var lock = LockService.getScriptLock();
    try {
      lock.waitLock(10000);

      if (type === 'Individu') {
        var userRole = String(userSession.role || '').trim();
        var isAmOrRm = (userRole === '1-RM' || userRole === '2-AM');
        var thresholdMin = isAmOrRm ? (8 * 60) : (7 * 60 + 45);
        
        var finalStatus = inputStatus;
        if (inputStatus === 'Hadir' && totalMinutes > thresholdMin) {
          finalStatus = 'Terlambat';
        }

        var absenId = 'ABS-' + Date.now() + '-' + Math.floor(100 + Math.random() * 900);
        absensiSheet.appendRow([
          absenId, dateStr, timeStr, userSession.nik, userSession.name, userSession.role,
          userSession.area, selectedPoint, 'Individu', finalStatus, remarks,
          userLat, userLng, distanceMeters, photoUrl, createdBy
        ]);

        absensiRecords.push({ name: userSession.name, status: finalStatus, keterangan: remarks });

      } else if (type === 'Kelompok') {
        var groupList = payload.groupList || [];
        if (!Array.isArray(groupList) || groupList.length === 0) {
          return { status: 'error', message: 'Daftar anggota kelompok tidak boleh kosong!' };
        }

        var allPointMembers = getBPListByPoint(selectedPoint);
        var checkedNiksMap = {};

        groupList.forEach(function(member) {
          var mNik = String(member.nik).trim();
          checkedNiksMap[mNik] = true;

          var isMemberAlreadyAbsen = false;
          for (var r = 1; r < absensiData.length; r++) {
            if (String(absensiData[r][3]).trim() === mNik && isSameDay_(absensiData[r][1], now)) {
              isMemberAlreadyAbsen = true;
              break;
            }
          }

          if (!isMemberAlreadyAbsen) {
            var memberRole = String(member.role || '4-BP').trim();
            var isMemberAmOrRm = (memberRole === '1-RM' || memberRole === '2-AM');
            var thresholdMin = isMemberAmOrRm ? (8 * 60) : (7 * 60 + 45);
            var mInputStatus = member.statusAbsen || 'Hadir';
            
            var mFinalStatus = mInputStatus;
            if (mInputStatus === 'Hadir' && totalMinutes > thresholdMin) {
              mFinalStatus = 'Terlambat';
            }

            var mRemarks = member.keterangan || remarks;
            var absenId = 'ABS-' + Date.now() + '-' + Math.floor(100 + Math.random() * 900);
            
            absensiSheet.appendRow([
              absenId, dateStr, timeStr, member.nik, member.name, memberRole,
              userSession.area, selectedPoint, 'Kelompok', mFinalStatus, mRemarks,
              userLat, userLng, distanceMeters, photoUrl, createdBy
            ]);

            absensiRecords.push({ name: member.name, status: mFinalStatus, keterangan: mRemarks });
          }
        });

        allPointMembers.forEach(function(bp) {
          var bpNik = String(bp.nik).trim();
          if (!checkedNiksMap[bpNik]) {
            var excludedKey = 'excl_group_' + bpNik + '_' + dateStr.replace(/\//g, '');
            cache.put(excludedKey, 'true', 86400);
          }
        });
      }

      SpreadsheetApp.flush();
      clearSheetCache_('absensi');
    } finally {
      lock.releaseLock();
    }

    if (userLat && userLng) {
      updateHeartbeat(userSession, userLat, userLng);
    }

    var waText = formatAbsensiWaText_({ point: selectedPoint, records: absensiRecords });
    return { status: 'success', message: 'Data Absensi ' + type + ' berhasil disimpan!', waText: waText, photoUrl: photoUrl };
  } catch (err) { return { status: 'error', message: err.toString() }; }
}

function formatAbsensiWaText_(data) {
  var lines = [];
  lines.push('*Laporan Absensi Regional Gorontalo*');
  lines.push('Point : ' + data.point);
  lines.push('');
  lines.push('Anggota :');
  data.records.forEach(function(rec) {
    var ketStr = rec.keterangan ? ' (' + rec.keterangan + ')' : '';
    lines.push('• ' + rec.name + ' - ' + rec.status + ketStr);
  });
  return lines.join('\n');
}

function getAbsensiSummaryReport(userSession, startDate, endDate, selectedMonth, filterArea, filterPoint, filterBP) {
  try {
    var rawResult = getRawDataWithHeaders_('absensi');
    var values = rawResult.rows || [];
    if (values.length === 0) return [];
    
    var role = userSession ? String(userSession.role || '').trim() : '';
    var userNik = userSession ? String(userSession.nik || '').trim() : '';
    var userPoint = userSession ? String(userSession.point || '').trim().toLowerCase() : '';
    var userArea = userSession ? String(userSession.area || '').trim().toLowerCase() : '';

    var dFromObj = startDate ? parseDateAny_(startDate) : null;
    var dToObj = endDate ? parseDateAny_(endDate) : null;
    if (dFromObj) dFromObj.setHours(0,0,0,0);
    if (dToObj) dToObj.setHours(23,59,59,999);

    var targetMonthNum = null;
    if (selectedMonth !== undefined && selectedMonth !== null && selectedMonth !== '' && selectedMonth !== 'all') {
      targetMonthNum = parseInt(selectedMonth, 10);
    }

    var fArea = String(filterArea || '').trim().toLowerCase();
    var fPoint = String(filterPoint || '').trim().toLowerCase();
    var fBP = String(filterBP || '').trim().toLowerCase();

    var bpMap = {};

    values.forEach(function(r) {
      var rowAbsenId = r['_raw_0'] || r.absenid || '';
      var rowDateStr = r['_raw_1'] || r.tanggal || '';
      var rowNik = String(r['_raw_3'] || r.nik || '').trim();
      var rowName = String(r['_raw_4'] || r.nama || '').trim();
      var rowRole = String(r['_raw_5'] || r.role || '').trim();
      var rowArea = String(r['_raw_6'] || r.area || '').trim();
      var rowPoint = String(r['_raw_7'] || r.point || '').trim();
      var status = String(r['_raw_9'] || r.status || '').trim();
      var rowPhoto = cleanGoogleDriveUrl_(String(r['_raw_14'] || r.fotourl || r.foto || '').trim());

      if (role === '2-AM' && userArea && rowArea.toLowerCase() !== userArea) return;
      if (role === '3-BM' && userPoint && rowPoint.toLowerCase() !== userPoint) return;
      if ((role === '4-BP' || role === '5-BPR' || role === '6-BP Buffer') && userNik && rowNik !== userNik) return;

      if (fArea && rowArea.toLowerCase() !== fArea) return;
      if (fPoint && rowPoint.toLowerCase() !== fPoint) return;
      if (fBP && (rowName.toLowerCase().indexOf(fBP) === -1 && rowNik.indexOf(fBP) === -1)) return;

      var rowD = parseDateAny_(rowDateStr);

      if (targetMonthNum !== null && !isNaN(targetMonthNum) && targetMonthNum > 0) {
        if (!rowD || (rowD.getMonth() + 1) !== targetMonthNum) return;
      }

      if (dFromObj || dToObj) {
        if (rowD) {
          rowD.setHours(12,0,0,0);
          if (dFromObj && rowD < dFromObj) return;
          if (dToObj && rowD > dToObj) return;
        }
      }

      var key = rowNik || rowName;
      if (!key) return;

      var canViewPhoto = false;
      if (role === '1-RM' || role === 'Editor' || role === 'Admin' || role === '2-AM') {
        canViewPhoto = true;
      } else if (role === '3-BM') {
        canViewPhoto = (rowPoint.toLowerCase() === userPoint);
      } else if (role === '4-BP' || role === '5-BPR' || role === '6-BP Buffer') {
        canViewPhoto = (rowNik === userNik);
      }

      if (!bpMap[key]) {
        bpMap[key] = {
          absenId: rowAbsenId, 
          nik: rowNik, 
          name: rowName, 
          role: rowRole,
          point: rowPoint, 
          area: rowArea, 
          hadir: 0, 
          terlambat: 0, 
          izin: 0, 
          sakit: 0, 
          alpha: 0, 
          total: 0,
          latestPhoto: (canViewPhoto && rowPhoto && rowPhoto !== '-') ? rowPhoto : '',
          canViewPhoto: canViewPhoto
        };
      }

      if (status === 'Hadir') bpMap[key].hadir++;
      else if (status === 'Terlambat') bpMap[key].terlambat++;
      else if (status === 'Izin') bpMap[key].izin++;
      else if (status === 'Sakit') bpMap[key].sakit++;
      else if (status === 'Tanpa Keterangan') bpMap[key].alpha++;

      if (canViewPhoto && rowPhoto && rowPhoto !== '-') {
        bpMap[key].latestPhoto = rowPhoto;
      }

      bpMap[key].total++;
    });

    var summaryList = [];
    Object.keys(bpMap).forEach(function(k) { summaryList.push(bpMap[k]); });
    return summaryList;
  } catch (err) { return []; }
}

function deleteAbsensiRecord(payload, userSession) {
  try {
    if (!userSession || !userSession.role) return { status: 'error', message: 'Akses ditolak! Silakan login.' };

    var role = String(userSession.role).trim();
    if (role !== '1-RM' && role !== '2-AM' && role !== 'Editor' && role !== 'Admin') {

    if (!ss) return { status: 'error', message: 'Spreadsheet tidak aktif!' };

    if (!sheet || sheet.getLastRow() < 2) return { status: 'error', message: 'Sheet Absensi tidak ditemukan atau kosong.' };

    payload = payload || {};
    var targetNik = String(payload.nik || '').trim();
    var targetAbsenId = String(payload.absenId || payload.absen_id || '').trim();

    var lock = LockService.getScriptLock();
    try {
      lock.waitLock(5000);
      var values = sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn()).getDisplayValues();

      for (var i = values.length - 1; i >= 0; i--) {
        var rowAbsenId = String(values[i][0] || '').trim();
        var rowNik = String(values[i][3] || '').trim();

        var isMatch = false;
        if (targetAbsenId && rowAbsenId === targetAbsenId) isMatch = true;
        else if (targetNik && rowNik === targetNik) isMatch = true;

        if (isMatch) {
          sheet.deleteRow(i + 2);
          SpreadsheetApp.flush();
          clearSheetCache_('absensi');
          return { status: 'success', message: 'Data absensi personil berhasil dihapus dari spreadsheet!' };
        }
      }
    } finally {
      lock.releaseLock();
    }

    return { status: 'error', message: 'Data absensi tidak ditemukan.' };
  } catch (err) { return { status: 'error', message: err.toString() }; }
}

function getDailyBPSummary(userSession, targetDateStr) {
  try {
    var now = new Date();
    var targetDateObj = targetDateStr ? parseDateAny_(targetDateStr) : now;
    if (!targetDateObj || isNaN(targetDateObj.getTime())) targetDateObj = now;
    
    var role = userSession ? String(userSession.role || '').trim() : '';
    var userNik = userSession ? String(userSession.nik || '').trim() : '';
    var userName = userSession ? String(userSession.name || '').trim().toLowerCase() : '';
    var userPoint = userSession ? String(userSession.point || '').trim().toLowerCase() : '';
    var userArea = userSession ? String(userSession.area || '').trim().toLowerCase() : '';
    var isBpRole = (role === '4-BP' || role === '5-BPR' || role === '6-BP Buffer');

    var rawHasil = getRawDataWithHeaders_('dashboard');
    var rawSos = getRawDataWithHeaders_('sosialisasi');
    var rawPel = getRawDataWithHeaders_('pelayanan');
    
    var hasilRows = rawHasil.rows || [];
    var sosRows = rawSos.rows || [];
    var pelRows = rawPel.rows || [];
    
    var bpRecords = [];

    var isRowPermitted = function(rArea, rPoint, rNik, rBpName) {
      if (role === '1-RM' || role === 'Editor' || role === 'Admin') return true;
      if (role === '2-AM') return rArea.toLowerCase() === userArea;
      if (role === '3-BM') return rPoint.toLowerCase() === userPoint;
      if (isBpRole) {
        if (userNik && rNik && rNik === userNik) return true;
        if (userName && rBpName && calculateNameSimilarity_(rBpName, userName) >= 0.90) return true;
        return false;
      }
      return true;
    };

    // 1. Sinkronisasi Data Sosialisasi (Aman vs Potensi Palsu)
    sosRows.forEach(function(r) {
      var rDate = r.tanggal || r.tanggalapproveuk || r.submittedat || r.waktu || r['_raw_0'] || '';
      if (!isSameDay_(rDate, targetDateObj)) return;

      var bpName = String(r.bpbm || r.namabp || r.bp || r.nama || r['_raw_2'] || r['_raw_1'] || '').trim();
      var point = String(r.point || r.branch || r['_raw_3'] || '').trim();
      var area = String(r.area || r['_raw_10'] || r['_raw_4'] || '').trim();
      var rNik = String(r.nik || r.nikbp || r['_raw_1'] || '').trim();

      if (!isRowPermitted(area, point, rNik, bpName)) return;

      var ket = String(r.keterangan || r.proses || r.hasilvisit || r['_raw_12'] || r['_raw_7'] || '').toLowerCase();
      var isPalsu = (ket.indexOf('palsu') !== -1 || ket.indexOf('potensi palsu') !== -1 || ket.indexOf('potensi') !== -1);
      
      findOrMergeBPRecord_(bpRecords, bpName, point, area, rNik, function(rec) {
        rec.sosTotal++;
        if (isPalsu) rec.sosPalsu++; else rec.sosAman++;
      });
    });

    // 2. Sinkronisasi Data Pelayanan (D2D, Kumpul, Semi)
    pelRows.forEach(function(r) {
      var rDate = r.tanggal || r.waktu || r['_raw_8'] || r['_raw_0'] || '';
      if (!isSameDay_(rDate, targetDateObj)) return;

      var bpName = String(r.bpbm || r['nama bp/bm'] || r.bp || r.nama || r['_raw_2'] || '').trim();
      var point = String(r.point || r.branch || r['_raw_1'] || '').trim();
      var area = String(r.area || r['_raw_0'] || '').trim();
      var rNik = String(r.nik || '').trim();

      if (!isRowPermitted(area, point, rNik, bpName)) return;

      var jkm = String(r.jeniskumpulanmajelis || r['_raw_4'] || '').toLowerCase();
      
      findOrMergeBPRecord_(bpRecords, bpName, point, area, rNik, function(rec) {
        rec.pelayananTotal++;
        if (jkm.indexOf('semi') !== -1) {
          rec.pelayananSemi++;
        } else if (jkm.indexOf('dor') !== -1 || jkm.indexOf('door') !== -1 || jkm.indexOf('d2d') !== -1) {
          rec.pelayananD2D++;
        } else {
          rec.pelayananKumpul++;
        }
      });
    });

    // 3. Sinkronisasi Data Kunjungan Eligible, DO, dan Penagihan Kolek
    hasilRows.forEach(function(r) {
      if (r.is_sosialisasi) return; // Lewati baris hasil mapping Sosialisasi di dashboard
      var rDate = r.tanggal || r.waktu || r['_raw_12'] || r['_raw_0'] || '';
      if (!isSameDay_(rDate, targetDateObj)) return;

      var bpName = String(r.bpbm || r['nama bp/bm'] || r.bp || r.nama || r['_raw_1'] || '').trim();
      var point = String(r.point || r.branch || r['_raw_2'] || '').trim();
      var area = String(r.area || r['_raw_3'] || '').trim();
      var rNik = String(r.nik || r['_raw_0'] || '').trim();

      if (!isRowPermitted(area, point, rNik, bpName)) return;

      var act = String(r.aktivitas || r['_raw_4'] || '').toLowerCase();
      var hst = String(r.hasilvisit || r.keterangan || r['_raw_8'] || '').toLowerCase();
      var isMinat = (hst.indexOf('minat') !== -1 && hst.indexOf('tidak') === -1 && hst.indexOf('belum') === -1);
      var collect = parseFloat(String(r.nominalcollect || r['_raw_10'] || '0').replace(/[^0-9]/g, '')) || 0;

      findOrMergeBPRecord_(bpRecords, bpName, point, area, rNik, function(rec) {
        if (act.indexOf('eligible') !== -1 || act.indexOf('wl') !== -1) {
          rec.wlTotal++;
          if (isMinat) rec.wlMinat++; else rec.wlTidakMinat++;
        } else if (act.indexOf('do') !== -1) {
          rec.doTotal++;
          if (isMinat) rec.doMinat++; else rec.doTidakMinat++;
        } else if (act.indexOf('sos') !== -1) {
          rec.sosTotal++;
          var isP = (hst.indexOf('palsu') !== -1 || hst.indexOf('potensi palsu') !== -1 || hst.indexOf('potensi') !== -1);
          if (isP) rec.sosPalsu++; else rec.sosAman++;
        } else if (act.indexOf('pelayanan') !== -1) {
          rec.pelayananTotal++;
          var jkm2 = String(r.jeniskumpulanmajelis || hst).toLowerCase();
          if (jkm2.indexOf('semi') !== -1) rec.pelayananSemi++;
          else if (jkm2.indexOf('dor') !== -1 || jkm2.indexOf('door') !== -1) rec.pelayananD2D++;
          else rec.pelayananKumpul++;
        } else {
          rec.kolekTotal++;
          rec.totalCollect += collect;
        }
      });
    });

    return bpRecords;
  } catch (e) {
    return [];
  }
}

function findOrMergeBPRecord_(list, bpName, point, area, nik, callback) {
  if (!bpName) return;
  var matchedRec = null;

  for (var i = 0; i < list.length; i++) {
    var sim = calculateNameSimilarity_(list[i].bpName, bpName);
    if (sim >= 0.90) {
      matchedRec = list[i];
      break;
    }
  }

  if (!matchedRec) {
    matchedRec = {
      nik: nik || '',
      bpName: bpName,
      point: point || '-',
      area: area || '-',
      sosTotal: 0,
      sosAman: 0,
      sosPalsu: 0,
      pelayananTotal: 0,
      pelayananD2D: 0,
      pelayananKumpul: 0,
      pelayananSemi: 0,
      wlTotal: 0,
      wlMinat: 0,
      wlTidakMinat: 0,
      doTotal: 0,
      doMinat: 0,
      doTidakMinat: 0,
      kolekTotal: 0,
      totalCollect: 0
    };
    list.push(matchedRec);
  }

  callback(matchedRec);
}

function getApprovalMatrixSummary(userSession, startDate, endDate, selectedMonthKey, selectedWeekIndex, filterPoint, filterArea, filterBP) {
  try {
    var role = userSession ? String(userSession.role || '').trim() : '';
    var userNik = userSession ? String(userSession.nik || '').trim() : '';
    var userPoint = userSession ? String(userSession.point || '').trim().toLowerCase() : '';
    var userArea = userSession ? String(userSession.area || '').trim().toLowerCase() : '';
    var isBpRole = (role === '4-BP' || role === '5-BPR' || role === '6-BP Buffer');

    var mppSheet = findSheet_('Data MPP');
    var appSheet = findSheet_('Aproval Perhari') || findSheet_('KP Daily');
    if (!mppSheet || !appSheet) return { matrix: [], dates: [], allDates: [], months: [], selectedMonth: '', weeks: [], currentWeekIndex: 0 };

    var mppData = mppSheet.getDataRange().getDisplayValues();
    var appData = appSheet.getDataRange().getDisplayValues();

    var filterPt = String(filterPoint || '').trim().toLowerCase();
    var filterAr = String(filterArea || '').trim().toLowerCase();
    var filterBpQuery = String(filterBP || '').trim().toLowerCase();

    var bpMap = {};
    for (var m = 1; m < mppData.length; m++) {
      var pos = String(mppData[m][8] || '').trim();
      var isBp = (pos === '4-BP' || pos === '5-BPR' || pos === '6-BP Buffer');
      var nik = String(mppData[m][3] || '').trim();
      var name = String(mppData[m][4] || '').trim();
      var pt = String(mppData[m][2] || '').trim();
      var ar = String(mppData[m][1] || '').trim();

      if (role === '2-AM' && userArea && ar.toLowerCase() !== userArea) continue;
      if (role === '3-BM' && userPoint && pt.toLowerCase() !== userPoint) continue;
      if (isBpRole && userNik && nik !== userNik) continue;

      if (filterPt && pt.toLowerCase() !== filterPt) continue;
      if (filterAr && ar.toLowerCase() !== filterAr) continue;
      if (filterBpQuery && name.toLowerCase().indexOf(filterBpQuery) === -1 && nik.indexOf(filterBpQuery) === -1) continue;

      if (isBp && nik) {
        bpMap[nik] = {
          nik: nik,
          name: name,
          point: pt,
          area: ar,
          datesMap: {}
        };
      }
    }

    var headers = appData[0];
    var colDate = -1, colCust = -1, colNik = -1, colName = -1;

    for (var h = 0; h < headers.length; h++) {
      var hNorm = canonicalHeader_(headers[h]);
      if (hNorm === 'tanggal_approve_uk' || hNorm === 'tanggal') colDate = h;
      if (hNorm === 'customer_number') colCust = h;
      if (hNorm === 'nik') colNik = h;
      if (hNorm === 'bpbm') colName = h;
    }

    var monthNamesId = ['Januari', 'Februari', 'Maret', 'April', 'Mei', 'Juni', 'Juli', 'Agustus', 'September', 'Oktober', 'November', 'Desember'];
    var availableMonthsMap = {};
    var allValidDatesMap = {};
    var rawApprovals = [];

    for (var r = 1; r < appData.length; r++) {
      var row = appData[r];
      var rDate = (colDate > -1 ? row[colDate] : row[11]) || '';
      var rCust = (colCust > -1 ? row[colCust] : row[3]) || '';
      var rNik = (colNik > -1 ? row[colNik] : row[12]) || '';
      var rName = (colName > -1 ? row[colName] : row[13]) || '';

      if (!rDate || !rCust) continue;

      var dObj = parseDateAny_(rDate);
      if (!dObj || isNaN(dObj.getTime())) continue;
      if (dObj.getDay() === 0) continue;

      var mKey = dObj.getFullYear() + '-' + ('0' + (dObj.getMonth() + 1)).slice(-2);
      var mLabel = monthNamesId[dObj.getMonth()] + ' ' + dObj.getFullYear();
      availableMonthsMap[mKey] = mLabel;

      var dateKey = ('0' + dObj.getDate()).slice(-2) + '/' + ('0' + (dObj.getMonth() + 1)).slice(-2) + '/' + dObj.getFullYear();
      allValidDatesMap[dateKey] = { dObj: dObj, time: dObj.getTime(), monthKey: mKey };

      rawApprovals.push({
        dateKey: dateKey,
        monthKey: mKey,
        rCust: String(rCust).trim(),
        rNik: String(rNik).trim(),
        rName: String(rName).trim()
      });
    }

    var monthsList = Object.keys(availableMonthsMap).sort().map(function(k) {
      return { key: k, label: availableMonthsMap[k] };
    });

    var now = new Date();
    var currentMKey = Utilities.formatDate(now, CONFIG.TIMEZONE, 'yyyy-MM');
    var activeMonthKey = selectedMonthKey;

    if (!activeMonthKey || !availableMonthsMap[activeMonthKey]) {
      if (availableMonthsMap[currentMKey]) {
        activeMonthKey = currentMKey;
      } else {
        activeMonthKey = monthsList.length > 0 ? monthsList[monthsList.length - 1].key : currentMKey;
      }
    }

    var monthDatesKeys = Object.keys(allValidDatesMap).filter(function(dKey) {
      return allValidDatesMap[dKey].monthKey === activeMonthKey;
    }).sort(function(a, b) {
      return allValidDatesMap[a].time - allValidDatesMap[b].time;
    });

    var weeks = [];
    var currentChunk = [];

    for (var i = 0; i < monthDatesKeys.length; i++) {
      var dtStr = monthDatesKeys[i];
      var dtObj = allValidDatesMap[dtStr].dObj;

      if (dtObj.getDay() === 1 && currentChunk.length > 0) {
        weeks.push({
          index: weeks.length,
          label: 'Minggu ke-' + (weeks.length + 1) + ' (' + currentChunk[0] + ' - ' + currentChunk[currentChunk.length - 1] + ')',
          dates: currentChunk
        });
        currentChunk = [];
      }

      currentChunk.push(dtStr);

      if (currentChunk.length >= 6) {
        weeks.push({
          index: weeks.length,
          label: 'Minggu ke-' + (weeks.length + 1) + ' (' + currentChunk[0] + ' - ' + currentChunk[currentChunk.length - 1] + ')',
          dates: currentChunk
        });
        currentChunk = [];
      }
    }

    if (currentChunk.length > 0) {
      weeks.push({
        index: weeks.length,
        label: 'Minggu ke-' + (weeks.length + 1) + ' (' + currentChunk[0] + ' - ' + currentChunk[currentChunk.length - 1] + ')',
        dates: currentChunk
      });
    }

    var nowTs = now.getTime();
    var detectedWeekIdx = 0;
    if (activeMonthKey === currentMKey) {
      for (var wk = 0; wk < weeks.length; wk++) {
        var wObj = weeks[wk];
        var startTs = allValidDatesMap[wObj.dates[0]].time;
        var endTs = allValidDatesMap[wObj.dates[wObj.dates.length - 1]].time;
        if (nowTs >= startTs && nowTs <= (endTs + 86400000)) {
          detectedWeekIdx = wk;
          break;
        }
        if (wk === weeks.length - 1 && nowTs > endTs) {
          detectedWeekIdx = wk;
        }
      }
    }

    var activeWeekIdx = (selectedWeekIndex !== undefined && selectedWeekIndex !== null) ? parseInt(selectedWeekIndex, 10) : detectedWeekIdx;
    if (activeWeekIdx >= weeks.length) activeWeekIdx = Math.max(0, weeks.length - 1);

    var displayDates = (weeks[activeWeekIdx] && weeks[activeWeekIdx].dates) ? weeks[activeWeekIdx].dates : monthDatesKeys.slice(0, 6);

    var custTracker = {};
    rawApprovals.forEach(function(item) {
      if (item.monthKey !== activeMonthKey) return;
      if (displayDates.indexOf(item.dateKey) === -1) return;

      var targetBP = bpMap[item.rNik];
      if (!targetBP && item.rName) {
        for (var bNik in bpMap) {
          if (calculateNameSimilarity_(bpMap[bNik].name, item.rName) >= 0.90) {
            targetBP = bpMap[bNik];
            break;
          }
        }
      }

      if (targetBP) {
        var dedupKey = item.dateKey + '_' + targetBP.nik + '_' + item.rCust;
        if (!custTracker[dedupKey]) {
          custTracker[dedupKey] = true;
          targetBP.datesMap[item.dateKey] = (targetBP.datesMap[item.dateKey] || 0) + 1;
        }
      }
    });

    var resultMatrix = [];
    for (var nikKey in bpMap) {
      var bpItem = bpMap[nikKey];
      var rowRecord = {
        nik: bpItem.nik,
        name: bpItem.name,
        point: bpItem.point,
        area: bpItem.area,
        dailyStatus: {},
        zonkCount: 0,
        uk1Count: 0,
        uk2Count: 0
      };

      displayDates.forEach(function(dt) {
        var count = bpItem.datesMap[dt] || 0;
        if (count === 0) {
          rowRecord.dailyStatus[dt] = 'ZONK';
          rowRecord.zonkCount++;
        } else if (count === 1) {
          rowRecord.dailyStatus[dt] = 'UK 1';
          rowRecord.uk1Count++;
        } else {
          rowRecord.dailyStatus[dt] = 'UK >= 2 (' + count + ')';
          rowRecord.uk2Count++;
        }
      });

      resultMatrix.push(rowRecord);
    }

    return {
      matrix: resultMatrix,
      dates: displayDates,
      allDates: monthDatesKeys,
      months: monthsList,
      selectedMonth: activeMonthKey,
      weeks: weeks,
      currentWeekIndex: activeWeekIdx
    };
  } catch (err) {
    return { matrix: [], dates: [], allDates: [], months: [], selectedMonth: '', weeks: [], currentWeekIndex: 0 };
  }
}

function getPerformaPencapaianSummary(userSession) {
  try {
    if (!userSession || !userSession.nik) {
      return { status: 'error', message: 'Silakan login terlebih dahulu!' };
    }

    var parseNum = function(v) {
      if (v === null || v === undefined) return 0;
      if (typeof v === 'number') return v;
      var str = String(v).replace(/[^-0-9.,]/g, '').trim();
      if (!str) return 0;
      if (str.indexOf(',') > -1 && str.indexOf('.') > -1) {
        str = str.replace(/\./g, '').replace(',', '.');
      } else if (str.indexOf(',') > -1) {
        str = str.replace(',', '.');
      }
      var num = parseFloat(str);
      return isNaN(num) ? 0 : num;
    };

    var pointRes = getRawDataWithHeaders_('pencapaianPoint');
    var pointRows = pointRes.rows || [];
    var pointList = [];

    pointRows.forEach(function(r) {
      var pt = r.point || r.branch || r['_raw_0'] || r['_raw_1'] || '';
      var ar = r.area || r['_raw_1'] || r['_raw_2'] || '';
      var achTotal = parseNum(r.achievmenttotal || r.achievementtotal || r['achtotal'] || r['_raw_14']);
      var totalDisb = parseNum(r.totalplafond || r.totaldisb || r.total || r['_raw_9'] || r['_raw_10']);
      var targetTotal = parseNum(r.targettotal || r.target || r['_raw_10'] || r['_raw_11']);
      var rank = parseInt(r.ranking || r.rank || r['_raw_16'] || r['_raw_15'], 10) || 999;

      if (pt && pt !== '-' && pt.toLowerCase() !== 'total') {
        pointList.push({
          point: pt,
          area: ar,
          achievmentTotal: achTotal,
          totalPlafond: totalDisb,
          targetTotal: targetTotal,
          ranking: rank
        });
      }
    });

    pointList.sort(function(a, b) {
      if (b.achievmentTotal !== a.achievmentTotal) {
        return b.achievmentTotal - a.achievmentTotal;
      }
      return a.ranking - b.ranking;
    });

    var top5Point = pointList.slice(0, 5);
    var bottom5Point = pointList.slice(Math.max(0, pointList.length - 5)).reverse();

    var bpRes = getRawDataWithHeaders_('pencapaian');
    var bpRows = bpRes.rows || [];
    var bpList = [];

    bpRows.forEach(function(r) {
      var name = r.bpbm || r.bp || r.nama || r.namabp || r['_raw_1'] || '';
      var nik = r.nik || r.nikbp || r['_raw_0'] || '';
      var pt = r.point || r.branch || r['_raw_2'] || '';
      var ar = r.area || r['_raw_3'] || '';
      var achTotal = parseNum(r.achievmenttotal || r.achievementtotal || r['achtotal'] || r['_raw_14']);
      var totalDisb = parseNum(r.totalplafond || r.totaldisb || r['_raw_10']);
      var targetTotal = parseNum(r.target || r.targettotal || r['_raw_11']);
      var rank = parseInt(r.ranking || r.rank || r['_raw_16'] || r['_raw_15'], 10) || 999;

      if (name && name !== '-' && name.toLowerCase() !== 'total') {
        bpList.push({
          nik: nik,
          name: name,
          point: pt,
          area: ar,
          achievmentTotal: achTotal,
          totalPlafond: totalDisb,
          targetTotal: targetTotal,
          ranking: rank
        });
      }
    });

    bpList.sort(function(a, b) {
      if (b.achievmentTotal !== a.achievmentTotal) {
        return b.achievmentTotal - a.achievmentTotal;
      }
      return a.ranking - b.ranking;
    });

    var top10BP = bpList.slice(0, 10);
    var bottom10BP = bpList.slice(Math.max(0, bpList.length - 10)).reverse();

    return {
      status: 'success',
      top5Point: top5Point,
      bottom5Point: bottom5Point,
      top10BP: top10BP,
      bottom10BP: bottom10BP
    };
  } catch (err) {
    return { status: 'error', message: err.toString() };
  }
}
